Additional M28/R parity check (existing unchanged parity/audit functions):
python3 -B - <<'PY'
import csv,hashlib,json,sys,time
from pathlib import Path
sys.path.insert(0,'scripts/tools')
from run_input import economic_parity
from run_demand import capture
from demand_audit import audit
from run_brewery import digest
from run_economic_diagnostics import write_json,write_rows
b=Path('/workspace/work/m29');traces=Path('/workspace/work/m28/traces');oldpath=Path('/workspace/work/m27/traces');seen={};rows=[]
pins={(r['condition'],r['days']):r for r in csv.DictReader(open('docs/M26_RUNS.csv')) if r['arm']=='requests_smoothed'}
stagger={r['condition']:r for r in csv.DictReader(open('docs/M26_STAGGERED_RUNS.csv')) if r['arm']=='requests_smoothed'}
for c,d,arm,fixture in [(c,d,'R_baseline_instrumented','none') for c in ['reference','tight','scarce','no_brew_request'] for d in [1,3,30]]+ [('reference',d,'R_input_aware','none') for d in [1,3,30]]+[(c,30,'R_baseline_instrumented','staggered') for c in ['tight','scarce']]:
 new=json.loads((traces/f'{c}-{d}-{arm}-{fixture}-1.json').read_text());key=c,d,fixture
 if key not in seen:
  path=oldpath/f'{c}-{d}-requests_smoothed.json'
  old=json.loads(path.read_text()) if fixture=='none' else capture(c,d,'requests_smoothed',fixture)
  audit(old);pin=pins[c,str(d)] if fixture=='none' else stagger[c]
  assert digest(dict(report=old['report']))==pin['summary_sha256'] and old['report']['ledger_sha256']==pin['ledger_sha256'];seen[key]=old
 rows.append(dict(condition=c,days=d,arm=arm,fixture=fixture,**economic_parity(new,seen[key]),M26_pins_unchanged=True))
 write_rows(b/'m28-economic-parity.csv',rows)
 print(c,d,arm,fixture,'M28/R full economic event/report/receipt parity PASS',flush=True)
assert len(rows)==17
